## C版本实现分析

### 1. GIOP层 (m70_giop.c)

**函数**: `melGetCurrentPrgBlock`

```c
long melGetCurrentPrgBlock(m70_conn_t* conn, int system_no, int row_count, void* msg)
{
    // 1. 构建GIOP header
    giop_header giop;
    build_giop_header(conn, &giop);
    
    // 2. 构建request header (operation_length = 0x1D = 29)
    request_pack_header request;
    build_request_pack_header(conn, &request, 0x1D);
    
    // 3. 构建prog_block_pack
    prog_block_pack pack;
    memset((void*)&pack, 0, sizeof(pack));
    
    pack.giop = giop;
    pack.request = request;
    strncpy(pack.op, op_command_get_prog_block, sizeof(pack.op));  // 32字节
    pack.op[sizeof(pack.op) - 1] = '\0';
    pack.principal = 0x00000000;
    pack.system_no = system_no;  // uint32
    pack.row_count = row_count;  // uint32
    
    // 4. 发送请求
    socket_send_data(conn->socket, &pack, sizeof(pack));
    
    // 5. 接收响应
    code = mel_receive_response(conn, &giop, &msg_length);
    if (code == 0) {
        msg_length -= receive_data_response(conn, msg_length, (int)T_STR, msg);
    }
}
```

m70_error_code_e m70_cnc_read_program_block(m70_conn_t* conn, short system_no, short row_count, char* msg_buff)

    long ret = melGetCurrentPrgBlock(conn, system_no, row_count, &msg);
    if (0 == ret) {
        int copy_len = msg.msg_length > 255 ? 255 : msg.msg_length;
        memcpy(msg_buff, msg.text, copy_len);

}

## Python版本实现检查

### 1. GIOP层 (m70_giop.py)

**函数**: `M70GIOP.mel_get_current_prg_block`

```python
@staticmethod
def mel_get_current_prg_block(conn: M70Connection, system_no: int, 
                               row_count: int) -> Tuple[int, Optional[bytes]]:
    # 1. 构建GIOP header
    giop_header = M70GIOP.build_giop_header(conn)
    
    # 2. 构建request header (0x1D = 29)
    request_header = M70GIOP.build_request_header(conn, 0x1D)
    
    # 3. 构建data packet
    packet = bytearray()
    
    # op字段: 32字节 ✓
    op_field = bytearray(32)
    op_bytes = M70GIOP.OP_GET_PROG_BLOCK + b'\x00'
    op_field[:len(op_bytes)] = op_bytes
    packet.extend(op_field)
    
    # 参数
    packet.extend(struct.pack('<I', 0))           # principal ✓
    packet.extend(struct.pack('<I', system_no))   # system_no (uint32) ✓
    packet.extend(struct.pack('<I', row_count))   # row_count (uint32) ✓
    
    # 4. 更新GIOP header并发送
    data_length = len(request_header) + len(packet)
    full_packet = bytearray(giop_header)
    struct.pack_into('<I', full_packet, 8, data_length)
    full_packet.extend(request_header)
    full_packet.extend(packet)
    
    M70Socket.send_data(conn._socket_obj, bytes(full_packet))
    
    # 5. 接收响应
    error_code, remaining_length = M70GIOP.receive_response(conn)
    if error_code != 0:
        return error_code, None
    
    # 6. 读取prog_block结构 ✓
    prog_block_data = None
    if remaining_length > 0:
        prog_block_data = M70Socket.recv_data(conn._socket_obj, remaining_length)
    
    return 0, prog_block_data
```

**常量**:
```python
OP_GET_PROG_BLOCK = b"mochaGetCurrentPrgBlockFirst"  # 29字节 + null = 30
```

### 2. 应用层 (m70_ezsocket.py)

**函数**: `M70Connection.read_program_block`

```python
def read_program_block(self, system_no: int = 1, row_count: int = 10) -> Tuple[M70ErrorCode, str]:
    # 1. 调用GIOP层方法
    error_code, prog_block_data = M70GIOP.mel_get_current_prg_block(self, system_no, row_count)
    
    if error_code != 0 or not prog_block_data:
        return M70ErrorCode.FAILED, ""
    
    # 2. 解析prog_block结构
    # Structure: int32 current_block, int32 current_row, int32 u1, int32 block_length, byte text[512]
    if len(prog_block_data) >= 16:
        current_block, current_row, u1_field, block_length = struct.unpack('<iiii', prog_block_data[0:16])
        
        if block_length > 0:
            text = prog_block_data[16:16+min(block_length, len(prog_block_data)-16)]
            result_text = text.decode('utf-8', errors='ignore').rstrip('\x00')
            return M70ErrorCode.OK, result_text
    
    return M70ErrorCode.FAILED, ""
```

## 关键细节说明

### 1. op字段大小的重要性

**错误配置 (16字节)**:
```python
op_field = bytearray(16)  # ❌ 太小！
```
结果: CNC返回error_code=2 (参数错误)

**正确配置 (32字节)**:
```python
op_field = bytearray(32)  # ✓ 正确！
```
结果: CNC正确响应，返回程序块数据

### 2. operation_length的计算

- 字符串: "mochaGetCurrentPrgBlockFirst" = 28字符
- 加上null terminator: 28 + 1 = 29 = 0x1D
- **不是**字符串长度 (28)
- **不是**op字段大小 (32)
- **是**字符串+null的长度 (29)

### 3. 响应解析方式

C版本:
```c
code = mel_receive_response(conn, &giop, &msg_length);
if (code == 0) {
    // 直接读取prog_block结构，无data header
    msg_length -= receive_data_response(conn, msg_length, (int)T_STR, msg);
}
```

Python版本:
```python
error_code, remaining_length = M70GIOP.receive_response(conn)
if error_code == 0:
    # 直接读取prog_block结构，无data header ✓

```
